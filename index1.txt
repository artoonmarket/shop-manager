// ── Google Sheets API ─────────────────────────────────────────────────────────

const BASE = 'https://sheets.googleapis.com/v4/spreadsheets';

async function sheetsGet(range) {
  const url = `${BASE}/${SHEET_ID}/values/${encodeURIComponent(range)}?key=${API_KEY}`;
  const r = await fetch(url);
  if (!r.ok) {
    const err = await r.json();
    throw new Error(err.error?.message || 'Sheets API error');
  }
  return r.json();
}

// Add this at the top of your script section
const BRIDGE_URL = 'https://script.google.com/macros/s/AKfycbzjYzVMrumms2u3LBskixUZktEUExrBEtmhRPNyHK-4Tu3FW6P5DyU3DkMewNmiGnJFXg/exec';

async function sheetsAppend(sheetName, values) {
  const r = await fetch(BRIDGE_URL, {
    method: 'POST',
    body: JSON.stringify({ action: 'append', sheet: sheetName, values: values[0] })
  });
  return r.json();
}

async function sheetsUpdate(range, values) {
  // Extract sheet name and cell range from the string (e.g., "Products!A1:F1")
  const [sheetName, cellRange] = range.split('!');
  const r = await fetch(BRIDGE_URL, {
    method: 'POST',
    body: JSON.stringify({ action: 'update', sheet: sheetName, range: cellRange, values: values })
  });
  return r.json();
}

async function sheetsClear(range) {
  const url = `${BASE}/${SHEET_ID}/values/${encodeURIComponent(range)}:clear?key=${API_KEY}`;
  const r = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' } });
  if (!r.ok) throw new Error('Clear failed');
  return r.json();
}

async function sheetsGetInfo() {
  const url = `${BASE}/${SHEET_ID}?key=${API_KEY}`;
  const r = await fetch(url);
  if (!r.ok) {
    const err = await r.json();
    throw new Error(err.error?.message || 'Cannot access spreadsheet');
  }
  return r.json();
}

async function ensureSheetsExist() {
  // Try to read info to validate creds
  const info = await sheetsGetInfo();
  const existingSheets = info.sheets.map(s => s.properties.title);

  const toAdd = [];
  if (!existingSheets.includes(PRODUCTS_SHEET)) toAdd.push(PRODUCTS_SHEET);
  if (!existingSheets.includes(SALES_SHEET)) toAdd.push(SALES_SHEET);

  if (toAdd.length > 0) {
    // We need to add sheets — requires OAuth, so we just try to read/write
    // and let the user know they need to create them manually
    toast(`Ստեղծեք «${toAdd.join('», «')}» թերթիկ(ներ) ձեր Spreadsheet-ում`, 'error');
    throw new Error(`Missing sheets: ${toAdd.join(', ')}`);
  }

  // Write headers if sheets are empty
  for (const sheet of [PRODUCTS_SHEET, SALES_SHEET]) {
    try {
      const data = await sheetsGet(`${sheet}!A1`);
      if (!data.values || !data.values[0] || !data.values[0][0]) {
        // Empty sheet — write headers
        if (sheet === PRODUCTS_SHEET) {
          await sheetsUpdate(`${sheet}!A1:F1`, [['id','name','quantity','purchasePrice','salePrice','addedAt']]);
        } else {
          await sheetsUpdate(`${sheet}!A1:G1`, [['id','productId','productName','quantity','salePrice','date','deleted']]);
        }
      }
    } catch(e) {
      // Sheet exists but maybe empty — write headers
      if (sheet === PRODUCTS_SHEET) {
        await sheetsUpdate(`${sheet}!A1:F1`, [['id','name','quantity','purchasePrice','salePrice','addedAt']]);
      } else {
        await sheetsUpdate(`${sheet}!A1:G1`, [['id','productId','productName','quantity','salePrice','date','deleted']]);
      }
    }
  }
}

// ── Data Loading ──────────────────────────────────────────────────────────────

async function loadData() {
  showLoading('Բեռնվում է...');
  try {
    const [prodData, salesData] = await Promise.all([
      sheetsGet(`${PRODUCTS_SHEET}!A2:F1000`),
      sheetsGet(`${SALES_SHEET}!A2:G1000`)
    ]);

    const pRows = prodData.values || [];
    state.products = pRows
      .filter(r => r[0] && r[1] && r[2] !== '' && r[3] !== '' && r[4] !== '')
      .map(r => ({
        id: r[0],
        name: r[1],
        quantity: Number(r[2]),
        purchasePrice: Number(r[3]),
        salePrice: Number(r[4]),
        addedAt: r[5] || ''
      }));

    const sRows = salesData.values || [];
    state.sales = sRows
      .filter(r => r[0] && r[6] !== '1') // skip deleted
      .map(r => ({
        id: r[0],
        productId: r[1],
        productName: r[2],
        quantity: Number(r[3]),
        salePrice: Number(r[4]),
        date: r[5] || ''
      }));

    render();
  } catch(e) {
    toast('Բեռնման սխալ: ' + e.message, 'error');
  }
  hideLoading();
}

// ── Sheets Write Helpers ──────────────────────────────────────────────────────

async function findRowInSheet(sheet, id) {
  // Reads all IDs from column A to find the row number
  const data = await sheetsGet(`${sheet}!A:A`);
  const rows = data.values || [];
  for (let i = 0; i < rows.length; i++) {
    if (rows[i][0] === String(id)) return i + 1; // 1-based
  }
  return -1;
}

async function addProduct() {
  const name = document.getElementById('p-name').value.trim();
  const quantity = document.getElementById('p-qty').value;
  const purchasePrice = document.getElementById('p-buy').value;
  const salePrice = document.getElementById('p-sell').value;

  if (!name || !quantity || !purchasePrice || !salePrice) {
    return toast('Լրացրեք բոլոր դաշտերը', 'error');
  }

  const btn = document.getElementById('add-product-btn');
  btn.disabled = true;
  btn.innerHTML = '<span class="spinner"></span> Պահվում...';

  try {
    const id = Date.now().toString();
    const addedAt = new Date().toISOString();
    await sheetsAppend(`${PRODUCTS_SHEET}!A:F`, [[id, name, Number(quantity), Number(purchasePrice), Number(salePrice), addedAt]]);

    toast('✅ Ապրանքն ավելացվեց!', 'success');
    document.getElementById('p-name').value = '';
    document.getElementById('p-qty').value = '';

  }

  btn.disabled = false;
  btn.innerHTML = '➕ Ավելացնել ապրանք';
}

async function deleteProduct(id) {
  if (!confirm('Ջնջե՞լ ապրանքը')) return;
  showLoading('Ջնջվում...');
  try {
    const rowNum = await findRowInSheet(PRODUCTS_SHEET, id);
    if (rowNum < 0) throw new Error('Ապրանք չի գտնվել');
    // Clear the row by setting all cells to empty
    await sheetsUpdate(`${PRODUCTS_SHEET}!A${rowNum}:F${rowNum}`, [['','','','','','']]);
    toast('🗑 Ջնջվեց', 'success');
    await loadData();
  } catch(e) {
    toast('Սխալ: ' + e.message, 'error');
    hideLoading();
  }
}

async function updateProductField(id, field, value) {
  // field: 'quantity'=col C (3), 'purchasePrice'=col D (4), 'salePrice'=col E (5)
  const colMap = { quantity: 'C', purchasePrice: 'D', salePrice: 'E' };
  const col = colMap[field];
  if (!col) return;

  showLoading('Թարմացվում...');
  try {
    const rowNum = await findRowInSheet(PRODUCTS_SHEET, id);
    if (rowNum < 0) throw new Error('Ապրանք չի գտնվել');
    await sheetsUpdate(`${PRODUCTS_SHEET}!${col}${rowNum}`, [[Number(value)]]);
    toast('✅ Թարմացվեց', 'success');
    await loadData();
  } catch(e) {
    toast('Սխալ: ' + e.message, 'error');
    hideLoading();
  }
}

async function recordSale() {
  const productId = document.getElementById('s-product').value;
  const qty = Number(document.getElementById('s-qty').value);
  const salePrice = Number(document.getElementById('s-price').value);

  if (!productId || !qty || !salePrice) return toast('Լրացրեք բոլոր դաշտերը', 'error');

  const product = state.products.find(p => p.id === productId);
  if (!product) return toast('Ապրանք չի գտնվել', 'error');
  if (product.quantity < qty) return toast('Պահեստում բավարար քանակ չկա', 'error');

  const btn = document.getElementById('sell-btn');
  btn.disabled = true;
  btn.innerHTML = '<span class="spinner"></span> Գրանցվում...';

  try {
    // 1. Append sale row
    const saleId = Date.now().toString();
    await sheetsAppend(`${SALES_SHEET}!A:G`, [[saleId, product.id, product.name, qty, salePrice, new Date().toISOString(), '0']]);

    // 2. Update product quantity
    const newQty = product.quantity - qty;
    const rowNum = await findRowInSheet(PRODUCTS_SHEET, product.id);
    if (rowNum > 0) {
      await sheetsUpdate(`${PRODUCTS_SHEET}!C${rowNum}`, [[newQty]]);
    }

    toast('✅ Վաճառքը գրանցվեց!', 'success');
    document.getElementById('s-product').value = '';
    document.getElementById('s-qty').value = '1';
    document.getElementById('s-price').value = '';
    document.getElementById('sell-stock-info').textContent = '';
    await loadData();
    populateSellDropdown();
  } catch(e) {
    toast('Սխալ: ' + e.message, 'error');
  }

  btn.disabled = false;
  btn.innerHTML = '✅ Հաստատել վաճառքը';
}

async function deleteSale(id) {
  if (!confirm('Ջնջե՞լ վաճառքը (քանակը կվերականգնվի)')) return;

  const sale = state.sales.find(s => s.id === id);
  if (!sale) return;

  showLoading('Ջնջվում...');
  try {
    // 1. Mark sale as deleted
    const rowNum = await findRowInSheet(SALES_SHEET, id);
    if (rowNum > 0) {
      await sheetsUpdate(`${SALES_SHEET}!G${rowNum}`, [['1']]);
    }

    // 2. Restore product quantity
    const product = state.products.find(p => p.id === sale.productId);
    if (product) {
      const pRow = await findRowInSheet(PRODUCTS_SHEET, product.id);
      if (pRow > 0) {
        await sheetsUpdate(`${PRODUCTS_SHEET}!C${pRow}`, [[product.quantity + sale.quantity]]);
      }
    }

    toast('🗑 Ջնջվեց', 'success');
    await loadData();
  } catch(e) {
    toast('Սխալ: ' + e.message, 'error');
    hideLoading();
  }
}

// ── Inline editing ────────────────────────────────────────────────────────────

function makeEditable(td, productId, field, currentVal) {
  td.innerHTML = `<input class="inline-edit" type="number" value="${currentVal}" min="0" />`;
  const inp = td.querySelector('input');
  inp.focus();
  inp.select();

  inp.addEventListener('blur', async () => {
    const newVal = inp.value.trim();
    if (newVal !== '' && Number(newVal) !== currentVal) {
      await updateProductField(productId, field, newVal);
    } else {
      render(); // restore
    }
  });
  inp.addEventListener('keydown', e => {
    if (e.key === 'Enter') inp.blur();
    if (e.key === 'Escape') { render(); }
  });
}

function fmt(n) { return Math.round(n).toLocaleString('hy-AM') + ' ֏'; }

function renderDashboard() {
  const now = new Date();
  const monthSales = state.sales.filter(s => {
    const d = new Date(s.date);
    return d.getMonth() === now.getMonth() && d.getFullYear() === now.getFullYear();
  });
  const monthIncome = monthSales.reduce((sum, s) => sum + s.salePrice * s.quantity, 0);
  const invested = state.products.reduce((sum, p) => sum + p.purchasePrice * p.quantity, 0);
  const expected = state.products.reduce((sum, p) => sum + p.salePrice * p.quantity, 0);

  document.getElementById('stat-products').textContent = state.products.length;
  document.getElementById('stat-invested').textContent = fmt(invested);
  document.getElementById('stat-expected').textContent = fmt(expected);
  document.getElementById('stat-month-income').textContent = fmt(monthIncome);

  const tbody = document.getElementById('dash-products-body');
  if (!state.products.length) {
    tbody.innerHTML = '<tr><td colspan="4" class="empty">Ապրանք չկա — ավելացրեք առաջինը! 📦</td></tr>';
    return;
  }
  tbody.innerHTML = state.products.map(p => `
    <tr>
      <td>${p.name}</td>
      <td><span class="qty-badge ${p.quantity > 2 ? 'qty-ok' : 'qty-low'}">${p.quantity}</span></td>
      <td class="price">${fmt(p.salePrice)}</td>
      <td class="income">${fmt(p.salePrice * p.quantity)}</td>
    </tr>
  `).join('');
}

function renderProducts() {

  if (!state.products.length) {
    tbody.innerHTML = '<tr><td colspan="8" class="empty">Ապրանք չկա 📦</td></tr>';
    return;
  }
  tbody.innerHTML = state.products.map((p, i) => `
    <tr>
      <td style="color:var(--muted)">${i+1}</td>
      <td><strong>${p.name}</strong></td>
      <td>
        <span class="qty-badge ${p.quantity > 2 ? 'qty-ok' : 'qty-low'} editable"
              onclick="makeEditable(this.parentElement, '${p.id}', 'quantity', ${p.quantity})"
              title="Սեղմեք խմբագրելու">${p.quantity}</span>
      </td>
      <td class="price editable" onclick="makeEditable(this, '${p.id}', 'purchasePrice', ${p.purchasePrice})" title="Սեղմեք խմբագրելու">${fmt(p.purchasePrice)}</td>
      <td>${fmt(p.purchasePrice * p.quantity)}</td>
      <td class="price editable" onclick="makeEditable(this, '${p.id}', 'salePrice', ${p.salePrice})" title="Սեղմեք խմբագրելու">${fmt(p.salePrice)}</td>
      <td class="income">${fmt(p.salePrice * p.quantity)}</td>
      <td><button class="btn btn-danger" onclick="deleteProduct('${p.id}')">🗑</button></td>
    </tr>
  `).join('');
}

function populateSellDropdown() {
  const sel = document.getElementById('s-product');
  sel.innerHTML = '<option value="">— Ընտրել ապրանք —</option>' +
    state.products.filter(p => p.quantity > 0).map(p =>
      `<option value="${p.id}" data-price="${p.salePrice}" data-qty="${p.quantity}">${p.name} (${p.quantity} հ.)</option>`
    ).join('');
  sel.onchange = () => {
    const opt = sel.options[sel.selectedIndex];
    if (opt.value) {
      document.getElementById('s-price').value = opt.dataset.price;
      document.getElementById('sell-stock-info').textContent = `Պահեստում կա: ${opt.dataset.qty} հ.`;
    } else {
      document.getElementById('sell-stock-info').textContent = '';
    }
  };
}

function populateFilters() {
  const years = [...new Set(state.sales.map(s => new Date(s.date).getFullYear()))].sort((a,b) => b-a);
  const ySel = document.getElementById('filter-year');
  const curYear = ySel.value;
  ySel.innerHTML = '<option value="">Բոլոր տարիները</option>' +
    years.map(y => `<option value="${y}" ${String(y) === curYear ? 'selected' : ''}>${y}</option>`).join('');

  const mSel = document.getElementById('filter-month');
  const curMonth = mSel.value;
  mSel.innerHTML = '<option value="">Բոլոր ամիսները</option>' +
    MONTHS_HY.map((m, i) => `<option value="${i}" ${String(i) === curMonth ? 'selected' : ''}>${m}</option>`).join('');
}

function renderSalesHistory() {
  const monthFilter = document.getElementById('filter-month')?.value;
  const yearFilter = document.getElementById('filter-year')?.value;

  let sales = [...state.sales].sort((a, b) => new Date(b.date) - new Date(a.date));
  if (monthFilter !== '' && monthFilter !== undefined)
    sales = sales.filter(s => new Date(s.date).getMonth() === Number(monthFilter));
  if (yearFilter)
    sales = sales.filter(s => new Date(s.date).getFullYear() === Number(yearFilter));

  if (sales.length > 0) {
    document.getElementById('month-summary').style.display = 'grid';
    const income = sales.reduce((sum, s) => sum + s.salePrice * s.quantity, 0);
    const topMap = {};
    sales.forEach(s => { topMap[s.productName] = (topMap[s.productName] || 0) + s.quantity; });
    const top = Object.entries(topMap).sort((a,b) => b[1]-a[1])[0];
    document.getElementById('ms-count').textContent = sales.length;
    document.getElementById('ms-income').textContent = fmt(income);
    document.getElementById('ms-top').textContent = top ? top[0] : '—';
  } else {
    document.getElementById('month-summary').style.display = 'none';
  }

  const tbody = document.getElementById('sales-body');
  if (!sales.length) {
    tbody.innerHTML = '<tr><td colspan="6" class="empty">Վաճառք չի գրանցվել 🗓️</td></tr>';
    return;
  }
  tbody.innerHTML = sales.map(s => `
    <tr>
      <td style="color:var(--muted)">${new Date(s.date).toLocaleDateString('hy-AM')}</td>
      <td>${s.productName}</td>
      <td>${s.quantity}</td>
      <td class="price">${fmt(s.salePrice)}</td>
      <td class="income">${fmt(s.salePrice * s.quantity)}</td>
      <td><button class="btn btn-danger" onclick="deleteSale('${s.id}')">🗑</button></td>
    </tr>
  `).join('');
}

// ── CSV Export ────────────────────────────────────────────────────────────────

function downloadCSV() {
  const escape = v => `"${String(v).replace(/"/g, '""')}"`;

  let csv = 'Ապրանքների ցուցակ\n';
  csv += 'Անվ.,Քան.,Ձ.Գ.,Ններ.Գ.,Վ.Գ.,Ակ.Եկ.\n';
  state.products.forEach(p => {
    csv += [escape(p.name), p.quantity, p.purchasePrice, p.purchasePrice * p.quantity, p.salePrice, p.salePrice * p.quantity].join(',') + '\n';
  });

  csv += '\nՎաճառքների պատմություն\n';
  csv += 'Ամսաթիվ,Ապրանք,Քան.,Գ./հ.,Ընդ.Եկ.\n';
  state.sales.forEach(s => {
    csv += [new Date(s.date).toLocaleDateString('hy-AM'), escape(s.productName), s.quantity, s.salePrice, s.salePrice * s.quantity].join(',') + '\n';
  });

  const blob = new Blob(['\uFEFF' + csv], { type: 'text/csv;charset=utf-8;' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url; a.download = `khanuyt-${new Date().toISOString().slice(0,10)}.csv`;
  a.click();
  URL.revokeObjectURL(url);
  toast('📄 CSV ներբեռնվեց', 'success');
}

// ── Toast ─────────────────────────────────────────────────────────────────────

let toastTimer;
function toast(msg, type = 'success') {
  const el = document.getElementById('toast');
  el.textContent = msg;
  el.className = 'show ' + type;
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => { el.className = ''; }, 3500);
}
